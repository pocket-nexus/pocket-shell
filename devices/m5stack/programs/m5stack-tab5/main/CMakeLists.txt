        pocket_display_esp_dpi

        pocket_worker_esp

target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)

# The launcher is Pocket Shell, compiled from this repository's shells/3ds
# (shells/3ds/shell, with the iOS 6 kit and the icons of ui/ at the
# repository root) against the host config pocket.host.config.ts by
# pocket_system's pocket_shell_compile_app (project_include.cmake).
# A .pocket the host contract admits replaces it.
set(POCKETJS_TAB5_SHELL_PACKAGE "" CACHE FILEPATH "Launcher .pocket to embed instead of shells/3ds/shell")
if(POCKETJS_TAB5_SHELL_PACKAGE)
    pocket_shell_embed_package(
        TARGET ${COMPONENT_LIB}
        NAME app
        PACKAGE "${POCKETJS_TAB5_SHELL_PACKAGE}"
        HOST_CONFIG "../pocket.host.config.ts")
else()
    pocket_shell_compile_app(
        TARGET ${COMPONENT_LIB}
        NAME app
        MANIFEST "../../../../../shells/3ds/shell/pocket.app.json"
        PROJECT_ROOT "../../../../../shells/3ds"
        HOST_CONFIG "../pocket.host.config.ts")
endif()

# Local apps the firmware installs at boot (pocket_system's bundled
# packages): each entry is an app directory below devices/m5stack,
# compiled against the same host config as the launcher. tab5_bundled.c
# lists them for main.c.
set(POCKETJS_TAB5_BUNDLED_APPS "apps/tab5-demo;apps/unit-swap" CACHE STRING "App directories below devices/m5stack to install at boot")
set(bundled_includes "")
set(bundled_entries "")
set(bundled_count 0)
foreach(app_dir IN LISTS POCKETJS_TAB5_BUNDLED_APPS)
    get_filename_component(app_leaf "${app_dir}" NAME)
    string(MAKE_C_IDENTIFIER "${app_leaf}" app_symbol)
    string(TOLOWER "bundled_${app_symbol}" bundled_name)
    pocket_shell_compile_app(
        TARGET ${COMPONENT_LIB}
        NAME ${bundled_name}
        MANIFEST "../../../${app_dir}/pocket.app.json"
        HOST_CONFIG "../pocket.host.config.ts")
    string(APPEND bundled_includes "#include \"pocketjs_package_${bundled_name}.h\"\n")
    string(APPEND bundled_entries "  bundled[${bundled_count}] = pocketjs_package_${bundled_name};\n")
    math(EXPR bundled_count "${bundled_count} + 1")
endforeach()
# Prebuilt packages the firmware serves from flash beside the apps above:
# each entry is a .pocket file compiled with this host config, such as an
# applet that `bun run tab5 <applet>` at the repository root builds from a
# pocket-nexus/applets checkout (POCKET_APPLETS_ROOT, default ../applets
# beside this repository).
set(POCKETJS_TAB5_BUNDLED_PACKAGES "" CACHE STRING "Prebuilt .pocket files to install at boot")
foreach(package_file IN LISTS POCKETJS_TAB5_BUNDLED_PACKAGES)
    get_filename_component(package_leaf "${package_file}" NAME_WE)
    string(MAKE_C_IDENTIFIER "${package_leaf}" package_symbol)
    string(TOLOWER "bundled_package_${package_symbol}" bundled_name)
    pocket_shell_embed_package(
        TARGET ${COMPONENT_LIB}
        NAME ${bundled_name}
        PACKAGE "${package_file}"
        HOST_CONFIG "../pocket.host.config.ts")
    string(APPEND bundled_includes "#include \"pocketjs_package_${bundled_name}.h\"\n")
    string(APPEND bundled_entries "  bundled[${bundled_count}] = pocketjs_package_${bundled_name};\n")
    math(EXPR bundled_count "${bundled_count} + 1")
endforeach()
if(bundled_count EQUAL 0)
    set(bundled_body "  *count = 0;\n  return NULL;\n")
else()
    set(bundled_body "  static pocketjs_system_package_t bundled[${bundled_count}];\n${bundled_entries}  *count = ${bundled_count};\n  return bundled;\n")
endif()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tab5_bundled.c" CONTENT
"/* Generated by main/CMakeLists.txt from POCKETJS_TAB5_BUNDLED_APPS and
 * POCKETJS_TAB5_BUNDLED_PACKAGES. */
#include \"tab5_bundled.h\"

#include <stddef.h>

${bundled_includes}
const pocketjs_system_package_t *tab5_bundled(size_t *count) {
${bundled_body}}
" @ONLY)
target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/tab5_bundled.c")
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
