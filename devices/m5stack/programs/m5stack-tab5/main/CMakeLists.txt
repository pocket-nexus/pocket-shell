idf_component_register(
    SRCS "main.c" "tab5_camera_media.c" "tab5_keyboard.c"
    REQUIRES
        esp_driver_dma
        esp_driver_i2c
        esp_driver_ppa
        esp_lcd
        esp_mm
        esp_timer
        pocket_display_esp_dpi
        pocket_guest_quickjs
        pocket_render_esp_ppa
        pocket_runner
        pocket_runtime
        pocket_audio
        pocket_camera
        pocket_device
        pocket_http_esp
        pocket_io_esp
        pocket_media
        pocket_mqtt_esp
        pocket_system
        pocket_worker_esp
)

target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)

# The launcher is Pocket Shell, compiled from the third_party/pocket-shell
# submodule (shells/3ds/shell, with the shared iOS 6 kit and icons of its
# vendor/applets submodule) against the host config pocket.host.config.ts.
# A package the host contract admits replaces it.
set(POCKETJS_TAB5_SHELL_PACKAGE "" CACHE FILEPATH "Launcher package to embed instead of third_party/pocket-shell")
if(POCKETJS_TAB5_SHELL_PACKAGE)
    pocketjs_embed_package(
        TARGET ${COMPONENT_LIB}
        NAME app
        PACKAGE "${POCKETJS_TAB5_SHELL_PACKAGE}"
        HOST_CONFIG "../pocket.host.config.ts")
else()
    pocketjs_compile_app(
        TARGET ${COMPONENT_LIB}
        NAME app
        MANIFEST "../../../third_party/pocket-shell/shells/3ds/shell/pocket.app.json"
        PROJECT_ROOT "../../../third_party/pocket-shell/shells/3ds"
        HOST_CONFIG "../pocket.host.config.ts")
endif()

# Local apps the firmware installs at boot (pocket_system's bundled
# packages): each entry is an app directory below the repository root,
# compiled against the same host config as the launcher. tab5_bundled.c
# lists them for main.c.
set(POCKETJS_TAB5_BUNDLED_APPS "apps/tab5-demo;apps/unit-swap" CACHE STRING "App directories below the repository root to install at boot")
set(bundled_includes "")
set(bundled_entries "")
set(bundled_count 0)
foreach(app_dir IN LISTS POCKETJS_TAB5_BUNDLED_APPS)
    get_filename_component(app_leaf "${app_dir}" NAME)
    string(MAKE_C_IDENTIFIER "${app_leaf}" app_symbol)
    string(TOLOWER "bundled_${app_symbol}" bundled_name)
    pocketjs_compile_app(
        TARGET ${COMPONENT_LIB}
        NAME ${bundled_name}
        MANIFEST "../../../${app_dir}/pocket.app.json"
        HOST_CONFIG "../pocket.host.config.ts"
        PROFILE_EXPORT_DIR "${CMAKE_CURRENT_BINARY_DIR}/pocketjs/bundled/profile")
    string(APPEND bundled_includes "#include \"pocketjs_package_${bundled_name}.h\"\n")
    string(APPEND bundled_entries
        "  bundled[${bundled_count}] = (pocketjs_system_bundled_t){\n"
        "      pocketjs_package_${bundled_name}.data, pocketjs_package_${bundled_name}.size};\n")
    math(EXPR bundled_count "${bundled_count} + 1")
endforeach()
# Prebuilt packages the firmware serves from flash beside the apps above:
# each entry is a .pocket file compiled with this host config, such as an
# applet that pocket-shell's `bun run tab5` builds from its vendor/applets
# (the applets need a newer PocketJS than third_party/pocketjs compiles).
set(POCKETJS_TAB5_BUNDLED_PACKAGES "" CACHE STRING "Prebuilt .pocket files to install at boot")
foreach(package_file IN LISTS POCKETJS_TAB5_BUNDLED_PACKAGES)
    get_filename_component(package_leaf "${package_file}" NAME_WE)
    string(MAKE_C_IDENTIFIER "${package_leaf}" package_symbol)
    string(TOLOWER "bundled_package_${package_symbol}" bundled_name)
    pocketjs_embed_package(
        TARGET ${COMPONENT_LIB}
        NAME ${bundled_name}
        PACKAGE "${package_file}"
        HOST_CONFIG "../pocket.host.config.ts"
        PROFILE_EXPORT_DIR "${CMAKE_CURRENT_BINARY_DIR}/pocketjs/bundled/profile")
    string(APPEND bundled_includes "#include \"pocketjs_package_${bundled_name}.h\"\n")
    string(APPEND bundled_entries
        "  bundled[${bundled_count}] = (pocketjs_system_bundled_t){\n"
        "      pocketjs_package_${bundled_name}.data, pocketjs_package_${bundled_name}.size};\n")
    math(EXPR bundled_count "${bundled_count} + 1")
endforeach()
if(bundled_count EQUAL 0)
    set(bundled_body "  *count = 0;\n  return NULL;\n")
else()
    set(bundled_body "  static pocketjs_system_bundled_t bundled[${bundled_count}];\n${bundled_entries}  *count = ${bundled_count};\n  return bundled;\n")
endif()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tab5_bundled.c" CONTENT
"/* Generated by main/CMakeLists.txt from POCKETJS_TAB5_BUNDLED_APPS. */
#include \"tab5_bundled.h\"

#include <stddef.h>

${bundled_includes}
const pocketjs_system_bundled_t *tab5_bundled(size_t *count) {
${bundled_body}}
" @ONLY)
target_sources(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/tab5_bundled.c")
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
